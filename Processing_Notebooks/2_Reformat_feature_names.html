cell_count_avg$relative_cc <- cell_count_avg$Average_cc/cell_count_avg[cell_count_avg$Metadata_Chemical== ctrl_group, ]$Average_cc

features <- data.frame(feature = colnames(test_chem_well[-c(1:4)])) %>%
  mutate(
    # ── Core structural parsing ──────────────────────────────────────────────
    module  = str_extract(feature, "^[^_]+"),
    region  = str_extract(feature, "(cells|cytoplasm|nuclei)$"),
    
    # ── Channel extraction ───────────────────────────────────────────────────
    # Correlation has two channels; everything else has at most one
    channel1 = case_when(
      module == "Correlation" ~
        str_match(feature,
          "^Correlation_[^_]+_(AGP|ER|DNA|Mito|RNA)_(AGP|ER|DNA|Mito|RNA)_"
        )[, 2],
      TRUE ~ str_extract(feature, "_(AGP|ER|DNA|Mito|RNA)(_|$)") %>%
               str_remove_all("_")
    ),
    channel2 = case_when(
      module == "Correlation" ~
        str_match(feature,
          "^Correlation_[^_]+_(AGP|ER|DNA|Mito|RNA)_(AGP|ER|DNA|Mito|RNA)_"
        )[, 3],
      TRUE ~ NA_character_
    ),
    
    # ── Specific measurement label ───────────────────────────────────────────
    specific = case_when(
      
      # Granularity: no meaningful sub-type, label by scale
      module == "Granularity" ~ "Granularity",
      
      # AreaShape: split into Size vs Shape 
      str_detect(feature, paste(c(
        "Area","BoundingBoxArea","EquivalentDiameter",
        "MajorAxisLength","MinorAxisLength",
        "MaxFeretDiameter","MinFeretDiameter",
        "MaximumRadius","MeanRadius","MedianRadius","Perimeter"
      ), collapse = "|")) ~ "Size",
      str_detect(feature, paste(c(
        "Compactness","Eccentricity","Extent",
        "FormFactor","Solidity","Orientation","EulerNumber"
      ), collapse = "|")) ~ "Shape",

      # Intensity subtypes
      str_detect(feature, "IntegratedIntensity") ~ "IntegratedIntensity",
      str_detect(feature, "MeanIntensity")        ~ "MeanIntensity",
      str_detect(feature, "MaxIntensity")         ~ "MaxIntensity",
      str_detect(feature, "MinIntensity")         ~ "MinIntensity",
      str_detect(feature, "StdIntensity")         ~ "StdIntensity",
      str_detect(feature, "MedianIntensity")      ~ "MedianIntensity",
      str_detect(feature, "MADIntensity")         ~ "MADIntensity",
      str_detect(feature, "UpperQuartileIntensity") ~ "UpperQuartileIntensity",
      str_detect(feature, "LowerQuartileIntensity") ~ "LowerQuartileIntensity",
      str_detect(feature, "MassDisplacement")     ~ "MassDisplacement",

      # Texture subtypes
      str_detect(feature, "Gabor")                ~ "Gabor",
      str_detect(feature, "AngularSecondMoment")  ~ "AngularSecondMoment",
      str_detect(feature, "InverseDifferenceMoment") ~ "InverseDifferenceMoment",
      str_detect(feature, "SumAverage")           ~ "SumAverage",
      str_detect(feature, "SumEntropy")           ~ "SumEntropy",
      str_detect(feature, "SumVariance")          ~ "SumVariance",
      str_detect(feature, "DifferenceEntropy")    ~ "DifferenceEntropy",
      str_detect(feature, "DifferenceVariance")   ~ "DifferenceVariance",
      str_detect(feature, "InfoMeas1")            ~ "InfoMeas1",
      str_detect(feature, "InfoMeas2")            ~ "InfoMeas2",
      str_detect(feature, "Texture_Entropy")      ~ "Entropy",
      str_detect(feature, "Texture_Variance")     ~ "Variance",
      str_detect(feature, "Texture_Contrast")     ~ "Contrast",
      str_detect(feature, "Texture_Correlation")  ~ "TextureCorrelation",

      # Radial distribution subtypes
      str_detect(feature, "FracAtD")   ~ "FractionAtDistance",
      str_detect(feature, "MeanFrac")  ~ "MeanFraction",
      str_detect(feature, "RadialCV")  ~ "RadialCV",

      # Colocalization subtypes (within Correlation module)
      str_detect(feature, "Correlation_Costes")     ~ "Costes",
      str_detect(feature, "Correlation_K_")         ~ "KCoefficient",
      str_detect(feature, "Correlation_Manders")    ~ "Manders",
      str_detect(feature, "Correlation_Overlap")    ~ "Overlap",
      str_detect(feature, "Correlation_RWC")        ~ "RWC",
      str_detect(feature, "Correlation_Correlation") ~ "PearsonCorrelation",

      # Spatial / neighbors
      str_detect(feature, "AngleBetween")       ~ "Spatial",
      str_detect(feature, "FirstClosest")        ~ "Spatial",
      str_detect(feature, "SecondClosest")       ~ "Spatial",
      str_detect(feature, "NumberOfNeighbors")   ~ "Spatial",
      str_detect(feature, "PercentTouching")     ~ "Spatial",

      # Position / location metadata
      str_detect(feature, "CenterMassIntensity") ~ "Position",
      str_detect(feature, "MaxIntensity_[XY]")   ~ "Position",
      str_detect(feature, "Center_[XY]")         ~ "Position",
      str_detect(feature, "BoundingBox")         ~ "Position",

      # Fallback
      TRUE ~ str_extract(feature, "(?<=_)[^_]+")
    ),
  )
